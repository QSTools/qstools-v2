(() => {
  function getFiberRoot() {
    const rootEl = document.getElementById("__next") || document.body;
    const key = Object.keys(rootEl).find(k => k.startsWith("__reactFiber$") || k.startsWith("__reactContainer$"));
    return rootEl[key];
  }
  function collect(fiber, seen, results) {
    if (!fiber || seen.has(fiber)) return;
    seen.add(fiber);
    try {
      const props = fiber.memoizedProps;
      if (props && typeof props === "object") {
        const keys = Object.keys(props);
        if (keys.some(k => /rate|cost|hours|profit|revenue|ready|burden|margin|labour|staff|overhead|pool|group|assign|recovery|basis|start|floor|ceiling|effect|lever|row_id|status|modelled|target|source|headline|capacity|gap|markup|true_cost|net_profit/i.test(k))) {
          const name = (fiber.type && (fiber.type.displayName || fiber.type.name)) || String(fiber.elementType || "unknown");
          results.push({ component: name, props });
        }
      }
    } catch (e) {}
    collect(fiber.child, seen, results);
    collect(fiber.sibling, seen, results);
  }
  const root = getFiberRoot();
  const results = [];
  collect(root, new WeakSet(), results);
  function safeStringify(obj) {
    const objSeen = new WeakSet();
    return JSON.stringify(obj, function (key, value) {
      if (typeof value === "function") return undefined;
      if (value instanceof Node) return "[DOM Node]";
      if (value && typeof value === "object") {
        if (objSeen.has(value)) return "[Circular]";
        objSeen.add(value);
        if (key.startsWith("__react") || key === "_owner" || key === "stateNode" || key === "return" || key === "child" || key === "sibling" || key === "alternate") {
          return undefined;
        }
      }
      return value;
    }, 2);
  }
  const text = safeStringify(results);
  const blob = new Blob([text], { type: "text/plain" });
  const url = URL.createObjectURL(blob);
  const a = document.createElement("a");
  const ts = new Date().toISOString().replace(/[:.]/g, "-");
  a.href = url;
  a.download = `PAGE_NAME_smoke_test_${ts}.txt`;
  document.body.appendChild(a);
  a.click();
  a.remove();
  console.log(`Captured ${results.length} components.`);
})();
