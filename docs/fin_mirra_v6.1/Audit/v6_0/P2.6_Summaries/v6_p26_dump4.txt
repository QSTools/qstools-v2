  34: function calculate_daily_open_hours(day) {
  35:   if (!day?.is_open) {
  36:     return {
  37:       daily_open_hours: 0,
  38:       invalid_time_range: false,
  39:     };
  40:   }
  41: 
  42:   const open_minutes = parse_time_to_minutes(day.open_time);
  43:   const close_minutes = parse_time_to_minutes(day.close_time);
  44:   const break_minutes = Math.max(to_number(day.break_minutes), 0);
  45: 
  46:   if (open_minutes === null || close_minutes === null) {
  47:     return {
  48:       daily_open_hours: 0,
  49:       invalid_time_range: true,
  50:     };
  51:   }
  52: 
  53:   const MINUTES_PER_DAY = 24 * 60;
  54:   let gross_minutes;
  55: 
  56:   if (close_minutes === open_minutes) {
  57:     // Same open/close time represents a full continuous 24-hour day
  58:     // (e.g. 24hr operation, 3-shift factory) - not a zero-length day.
  59:     gross_minutes = MINUTES_PER_DAY;
  60:   } else if (close_minutes < open_minutes) {
  61:     // Closing time is after midnight (e.g. 10:00 PM to 6:00 AM) -
  62:     // wrap around to the next day instead of treating it as invalid.
  63:     gross_minutes = close_minutes + MINUTES_PER_DAY - open_minutes;
  64:   } else {
  65:     gross_minutes = close_minutes - open_minutes;
  66:   }
  67: 
  68:   const net_minutes = Math.max(gross_minutes - break_minutes, 0);
  69: 
  70:   return {
  71:     daily_open_hours: round_number(net_minutes / 60),
  72:     invalid_time_range: false,
  73:   };
