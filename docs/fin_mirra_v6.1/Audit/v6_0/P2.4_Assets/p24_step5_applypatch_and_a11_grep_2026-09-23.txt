  param([string]$Path, [string]$Find, [string]$Replace, [string]$Label)
  $full = Join-Path (Get-Location).Path $Path
  $text = [System.IO.File]::ReadAllText($full)
  $first = $text.IndexOf($Find, [StringComparison]::Ordinal)
  if ($first -lt 0) { "NOT FOUND: $Label"; return }
  $second = $text.IndexOf($Find, $first + $Find.Length, [StringComparison]::Ordinal)
  if ($second -ge 0) { "AMBIGUOUS: $Label"; return }
  $nl = if ($text.Contains("`r`n")) { "`r`n" } else { "`n" }
  $Replace = $Replace -replace "`r?`n", $nl
  $new = $text.Substring(0, $first) + $Replace + $text.Substring($first + $Find.Length)
  [System.IO.File]::WriteAllText($full, $new, (New-Object System.Text.UTF8Encoding($false)))
  "Patched: $Label"
